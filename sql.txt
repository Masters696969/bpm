-- Optional seed balances for current year
INSERT INTO employee_leave_balances
(EmployeeID, LeaveTypeID, Year, TotalCredits, UsedCredits, RemainingCredits)
SELECT
  e.EmployeeID,
  lt.LeaveTypeID,
  YEAR(CURDATE()),
  lt.DefaultCredits,
  0,
  lt.DefaultCredits
FROM employee e
CROSS JOIN leave_types lt
WHERE NOT EXISTS (
    SELECT 1
    FROM employee_leave_balances lb
    WHERE lb.EmployeeID = e.EmployeeID
      AND lb.LeaveTypeID = lt.LeaveTypeID
      AND lb.Year = YEAR(CURDATE())
);

-- 17) ATTENDANCE SESSION

CREATE TABLE IF NOT EXISTS `attendance_session` (
  `SessionID` int(11) NOT NULL AUTO_INCREMENT,
  `EmployeeID` int(11) NOT NULL,
  `WorkDate` date NOT NULL,
  `AssignmentID` int(11) DEFAULT NULL,
  `Status` enum('OPEN','CLOSED') DEFAULT 'OPEN',
  `CreatedAt` datetime DEFAULT current_timestamp(),
  `ClosedAt` datetime DEFAULT NULL,
  PRIMARY KEY (`SessionID`),
  KEY `idx_as_employee` (`EmployeeID`),
  KEY `idx_as_workdate` (`WorkDate`),
  KEY `idx_as_assignment` (`AssignmentID`),
  KEY `idx_as_employee_workdate` (`EmployeeID`, `WorkDate`),
  CONSTRAINT `fk_as_employee`
    FOREIGN KEY (`EmployeeID`) REFERENCES `employee` (`EmployeeID`)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  CONSTRAINT `fk_as_assignment`
    FOREIGN KEY (`AssignmentID`) REFERENCES `roster_assignment` (`AssignmentID`)
    ON DELETE SET NULL
    ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `attendance_event` (
  `EventID` int(11) NOT NULL AUTO_INCREMENT,
  `SessionID` int(11) NOT NULL,
  `EventType` enum('TIME_IN','BREAK_OUT','BREAK_IN','TIME_OUT') NOT NULL,
  `EventTime` datetime NOT NULL DEFAULT current_timestamp(),
  `Latitude` decimal(10,7) DEFAULT NULL,
  `Longitude` decimal(10,7) DEFAULT NULL,
  `LocationID` int(11) DEFAULT NULL,
  `DistanceMeters` decimal(10,2) DEFAULT NULL,
  `GeoStatus` enum('IN_GEOFENCE','OUTSIDE_GEOFENCE') DEFAULT NULL,
  `FaceStatus` enum('MATCH','NO_MATCH') DEFAULT NULL,
  `FaceScore` decimal(5,2) DEFAULT NULL,
  `CreatedAt` datetime DEFAULT current_timestamp(),
  PRIMARY KEY (`EventID`),
  KEY `idx_ae_session` (`SessionID`),
  KEY `idx_ae_location` (`LocationID`),
  KEY `idx_ae_type_time` (`EventType`, `EventTime`),
  CONSTRAINT `fk_ae_session`
    FOREIGN KEY (`SessionID`) REFERENCES `attendance_session` (`SessionID`)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  CONSTRAINT `fk_ae_location`
    FOREIGN KEY (`LocationID`) REFERENCES `work_locations` (`LocationID`)
    ON DELETE SET NULL
    ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
